        function getElement(id) {

        function isEmptyValue(value) {
            if (value === null || value === undefined) return true;
            if (typeof value === 'string') {
                const trimmed = value.trim();
                if (!trimmed) return true;
                if (trimmed.toLowerCase() === 'sem dados') return true;
            }
            return false;
        }

        function displayValue(value, fallback = '-') {
            return isEmptyValue(value) ? fallback : value;
        }

        /**
         * Atualiza um elemento simples por ID e oculta o conteiner marcado com
         * `data-empty-field` quando nao ha conteudo util para exibir.
         */
        function setText(id, value) {
            const element = getElement(id);
            if (!element) return;

            const target = element.closest('[data-empty-field]') || element;
            if (isEmptyValue(value)) {
                element.textContent = '';
                target.classList.add('hidden');
                return;
            }

            element.textContent = value;
            target.classList.remove('hidden');
        }

        function setHtml(id, html) {
            const element = getElement(id);
            if (!element) return;
            element.innerHTML = html;
        }

        function escapeHtml(value) {
            return String(value ?? '').replace(/[&<>'"]/g, function (char) {
                return {
                    '&': '&amp;',
                    '<': '&lt;',
                    '>': '&gt;',
                    "'": '&#39;',
                    '"': '&quot;'
                }[char];
            });
        }

        /**
         * Escapa valores usados dentro de onclick inline para manter o popup seguro.
         * Entrada: texto simples que sera interpolado em string JavaScript.
         * Saida: string segura para atributos inline.
         */
        function escapeJsString(value) {
            return String(value ?? '')
                .replace(/\r?\n/g, ' ')
                .replace(/\\/g, '\\\\')
                .replace(/'/g, "\\'");
        }

        /**

        function toggleSummaryCard(trigger) {

                item.classList.toggle('is-active', isActive);
                item.classList.toggle('folder-active', isActive);

        function getInitials(name) {
            const cleaned = String(name || '').trim();
            if (!cleaned) return 'SD';

            const parts = cleaned.split(/\s+/).filter(Boolean);
            if (!parts.length) return 'SD';

            const first = parts[0].charAt(0);
            const second = parts.length > 1 ? parts[1].charAt(0) : (parts[0].charAt(1) || '');
            return (first + second).toUpperCase();
        }

        function isValidIconUrl(url) {

        /**
         * Gera o icone visual de instituicao com fallback por iniciais.

        function getInstitutionIconMarkup(name, url) {
            const safeName = escapeHtml(name || '');
            const initials = getInitials(name);
            const fallbackHtml = '<div class="inst-icon inst-fallback">' + escapeHtml(initials) + '</div>';

            if (isValidIconUrl(url)) {

        /**
         * Preenche seletores de filtro com opcoes derivadas dos dados ja carregados.
         * Mantem a selecao atual sempre que o valor ainda existir na lista.

        function populateSelect(id, options, selected) {
            const select = getElement(id);
            if (!select) return;

            const unique = Array.from(new Set((options || []).filter(Boolean))).sort();
            const html = ['<option value="">Todos</option>']
                .concat(unique.map(function (option) {

                    return '<option value="' + safeOption + '">' + safeOption + '</option>';
                }))
                .join('');

            select.innerHTML = html;
            if (selected && unique.includes(selected)) {
                select.value = selected;
            }
        }

        function formatPercentValue(value) {
            if (isEmptyValue(value)) return '';
            if (typeof value === 'string') return value;

            const numericValue = Number(value);
            if (Number.isNaN(numericValue)) return '';

            const pct = Math.abs(numericValue) <= 1 ? numericValue * 100 : numericValue;
            return pct.toLocaleString('pt-BR', {
                minimumFractionDigits: 1,
                maximumFractionDigits: 1
            }) + '%';
        }

        function getPortfolioStatusText(score, alerts) {
            let base = 'Carteira em analise';
            const status = score?.status;

            if (status === 'Crítico' || status === 'Critico') base = 'Carteira em risco';
            else if (status === 'Instável' || status === 'Instavel') base = 'Carteira instavel';
            else if (status === 'Saudável' || status === 'Saudavel') base = 'Carteira saudavel';

            const criticalAlert = Array.isArray(alerts)
                ? alerts.find(function (alert) { return alert?.type === 'Crítico' || alert?.type === 'Critico'; })
                : null;

            if (criticalAlert) {
                const asset = criticalAlert.asset || criticalAlert.symbol || '';
                base += asset ? ' com risco em ' + asset : ' com alerta critico';
            }

            return base;
        }

        function getImpactText(action) {
            if (action === 'Vender') return 'reduz risco da carteira';
            if (action === 'Revisar') return 'ajusta posicao para evitar perdas';
            return 'mantem disciplina da carteira';
        }

        /**
         * Determina a acao em destaque com base em alertas criticos e, na ausencia
         * deles, na pior rentabilidade do conjunto de acoes em carteira.
         */
        function getRecommendedAction(alerts, actions) {
            const alertList = Array.isArray(alerts) ? alerts : [];
            const actionList = Array.isArray(actions) ? actions : [];
            const criticalAlert = alertList.find(function (alert) {
                return alert?.type === 'Crítico' || alert?.type === 'Critico';
            });

            if (criticalAlert) {
                const asset = criticalAlert.asset || criticalAlert.symbol || 'Ativo';
                const reason = criticalAlert.message ? 'Motivo: ' + criticalAlert.message : 'Motivo: alerta critico';
                return {
                    action: 'Vender',
                    asset: asset,
                    reason: reason,
                    impact: getImpactText('Vender')
                };
            }

            const worstAction = actionList.reduce(function (worst, item) {
                if (!item) return worst;
                if (!worst) return item;
                return (item.rent ?? 0) < (worst.rent ?? 0) ? item : worst;
            }, null);

            if (worstAction) {
                const asset = worstAction.ticker || worstAction.name || 'Ativo';
                const rentText = displayValue(
                    worstAction.rendimentoPct || formatPercentValue(worstAction.rent ?? 0),
                    'indisponivel'
                );
                return {
                    action: 'Revisar',
                    asset: asset,
                    reason: 'Motivo: rentabilidade ' + rentText,
                    impact: getImpactText('Revisar')
                };
            }

            return {
                action: 'Revisar',
                asset: 'Carteira',
                reason: 'Motivo: sem alertas criticos',
                impact: getImpactText('Revisar')
            };
        }

        /**

        function renderAlertsCard(alerts) {
            const alertElement = getElement('alert-text');
            if (!alertElement) return;

            const alertList = Array.isArray(alerts) ? alerts.slice() : [];
            if (!alertList.length) {
                alertElement.textContent = 'Sem alertas criticos';
                return;
            }

            const order = { 'Crítico': 0, 'Critico': 0, 'Atenção': 1, 'Atencao': 1, 'Informativo': 2 };
            alertList.sort(function (a, b) {
                return (order[a.type] ?? 9) - (order[b.type] ?? 9);
            });

                const tag = escapeHtml(String(alert.type || 'Info').toUpperCase());
                const message = escapeHtml(displayValue(alert.message, 'Sem detalhes'));
                return '<span class="block">' + tag + ': ' + message + '</span>';
            });

            alertElement.innerHTML = lines.join('');
        }

        function renderInvestmentsCard(items) {
            const container = getElement('invest-list');
            if (!container) return;

            container.innerHTML = '';
            const limitedItems = Array.isArray(items) ? items.slice(0, 3) : [];

            if (!limitedItems.length) {
                container.innerHTML = '<div class="type-body-xs text-slate-500">Nenhum investimento encontrado.</div>';
                return;
            }

            limitedItems.forEach(function (item) {
                const rentClass = item.rentRaw < 0 ? 'text-red-400' : 'text-emerald-400';
                const recommendationClass = getRecommendationClass(item.recommendation);
                const iconHtml = getInstitutionIconMarkup(item.institution, item.institutionIcon);
                const itemLabel = displayValue(item.name, 'Fundos de investimento');
                const chartHtml = '<button type="button" class="chart-btn text-slate-300" onclick="showFeaturePopup(\'chart\', \'' + escapeJsString(itemLabel) + '\')" title="Grafico e analise" aria-label="Grafico e analise de ' + escapeHtml(itemLabel) + '">' + appConfig.chartSvg + '</button>';

                const row = document.createElement('div');
                row.className = 'flex items-center justify-between gap-3 bg-white/5 p-3 rounded-none';
                row.innerHTML = ''
                    + '<div class="flex items-center gap-3">'
                    + iconHtml
                    + '    <div>'
                    + '        <p class="type-body-strong text-xs text-slate-200">' + escapeHtml(displayValue(item.name, '-')) + '</p>'
                    + '        <p class="type-body-xs text-slate-400">' + escapeHtml(displayValue(item.institution, '-')) + '</p>'
                    + '    </div>'
                    + '</div>'
                    + '<div class="text-right">'
                    + '    <p class="type-value-sm text-xs hide-value">' + escapeHtml(displayValue(item.valorAtual, '-')) + '</p>'
                    + '    <p class="type-body-xs ' + rentClass + ' hide-value">' + escapeHtml(displayValue(item.rentPct, '-')) + '</p>'
                    + '    <span class="px-2 py-1 ' + recommendationClass + ' rounded-none type-action">' + escapeHtml(displayValue(item.recommendation, '-')) + '</span>'
                    + '</div>'
                    + '<div>' + chartHtml + '</div>';

                container.appendChild(row);
            });
        }

        /**

        function renderFilteredCards() {
            renderActionsTable(getFilteredActions());
            renderInvestmentsCard(getFilteredInvestments());
            renderPrevidenciaCard(getFilteredPrevidencias());
        }

        /**
         * Orquestra a atualizacao completa da tela.
         * Cada card recebe apenas o subconjunto de dados necessario para permanecer isolado.
         */
        function renderApp() {
            const data = appState.dashboardData || {};

            renderHeader(data.summary || {}, data.score || {}, data.alerts || [], data.profile || {});
            renderRecommendedActionCard(data.alerts || [], data.actions || []);
            renderAIAnalysisCard(data.generalAdvice || '');
            renderPortfolioSummaryCards(data.summary || {}, data.previdenciaInfo || {}, data.fundosTop || []);
            renderAlertsCard(data.alerts || []);
            renderOrdersCard(data.orders || {});
            renderSupportNotesCard(data.score || {}, data.tip || '');
            renderFilterControls(data);
            renderFilteredCards();
        }

        // ---------------------------------------------------------------------
        // User Actions and Data Loading
        // ---------------------------------------------------------------------

        function handleDashboardDataLoaded(data) {
            appState.dashboardData = data || {};
            renderApp();
            setDashboardLoading(false);
        }

        function handleDashboardLoadFailure() {
            setDashboardLoading(false);
            setText('alert-text', 'Erro ao carregar dados');
        }

        /**
         * Carrega o payload principal do dashboard via Apps Script.
         * Entrada: nenhuma. Saida: atualiza appState.dashboardData e dispara renderApp.
         */
        function loadDashboardData() {
            setDashboardLoading(true);
            if (typeof google === 'undefined' || !google.script?.run) {
                setDashboardLoading(false);
                return;
            }

            google.script.run
                .withSuccessHandler(handleDashboardDataLoaded)
                .withFailureHandler(handleDashboardLoadFailure)
                .getDashboardData();
        }

        /**
         * Exibe um retorno local para a Esquilo IA enquanto o recurso real ainda nao foi liberado.
         * Mantem o texto travado no card para nao ser sobrescrito nos proximos refreshes.
         */
        function showAiUnavailableState(triggerButton) {
            const responseContainer = getElement('ai-response');
            if (!responseContainer) return;

            appState.aiLocked = true;

            if (triggerButton) {
                triggerButton.disabled = false;
                triggerButton.classList.remove('opacity-50');
                if (triggerButton.id === 'ai-btn') {
                    triggerButton.innerHTML = 'Modo em preparo';
                }
            }

            responseContainer.classList.remove('hidden');
            responseContainer.innerHTML = 'Esquilo IA ainda esta aquecendo os modulos taticos. Por enquanto seguimos no manual do squad e esse suporte entra no proximo loadout.';
            showFeaturePopup('ai', 'Esquilo IA');
        }

        /**
         * Aciona o placeholder local da Esquilo IA sem chamar backend ou alterar o fluxo de negocio.
         */
        function requestStrategyAnalysis() {
            const button = getElement('strategy-btn');
            showAiUnavailableState(button);
        }

        /**
         * Mantem compatibilidade com o fluxo legado de analise tatica.
         * Reaproveita o mesmo placeholder local para nao depender de servicos externos.
         */
        function requestAIAnalysis() {
            const button = getElement('ai-btn');
            showAiUnavailableState(button);
        }

        /**
         * Simula o envio visual da ordem na interface atual.
         * Mantido sem alterar a regra existente de apenas desabilitar o botao.
         */
        function executeSwap() {
            const button = getElement('swap-btn');
            if (!button) return;

            button.innerHTML = 'ORDEM ENVIADA';
            button.disabled = true;
            button.classList.replace('bg-amber-500', 'bg-slate-800');
            button.classList.add('text-slate-500', 'cursor-not-allowed');
        }

        // ---------------------------------------------------------------------
        // UI Events
        // ---------------------------------------------------------------------

        /**
         * Mantem compatibilidade com interacoes legadas de expansao fora do grid principal.
         * Entrada: elemento clicado.
         * Saida: alterna apenas a classe visual folder-active.
         */
        function toggleFolder(element) {
            if (!element) return;
            element.classList.toggle('folder-active');
        }

        /**
         * Alterna o modo fantasma da interface.
         * Usa o body principal e o botao do header para sincronizar estado visual e protecao de dados.
         */
        function toggleGhostMode() {
            const body = getElement('main-body');
            const icon = getElement('ghost-icon');
            const button = getElement('ghost-toggle');
            if (!body) return;

            const active = body.classList.toggle('ghost-active');
            if (button) {
                button.classList.toggle('app-icon-button--active', active);
            }
            if (icon) {
                icon.classList.toggle('text-amber-300', active);
                icon.classList.toggle('text-slate-300', !active);
            }
        }

        function toggleActionDetails(button) {
            const row = button?.closest('tr');
            const detailsRow = row?.nextElementSibling;
            if (!row || !detailsRow || !detailsRow.classList.contains('action-detail-row')) return;

            detailsRow.classList.toggle('hidden');
            button.textContent = detailsRow.classList.contains('hidden') ? 'Detalhes' : 'Ocultar';
        }

        /**
         * Registra os eventos dos filtros uma unica vez.
         * Cada alteracao apenas atualiza o estado local e re-renderiza os cards dependentes.
         */
        function bindFilterEvents() {
            if (appState.filtersBound) return;
            appState.filtersBound = true;

            const actionsInst = getElement('acoes-filter-inst');
            const actionsRec = getElement('acoes-filter-rec');
            const investInst = getElement('invest-filter-inst');
            const prevInst = getElement('prev-filter-inst');

            if (actionsInst) {
                actionsInst.addEventListener('change', function (event) {
                    appState.filters.actionsInst = event.target.value;
                    renderFilteredCards();
                });
            }

            if (actionsRec) {
                actionsRec.addEventListener('change', function (event) {
                    appState.filters.actionsRec = event.target.value;
                    renderFilteredCards();
                });
            }

            if (investInst) {
                investInst.addEventListener('change', function (event) {
                    appState.filters.investInst = event.target.value;
                    renderFilteredCards();
                });
            }

            if (prevInst) {
                prevInst.addEventListener('change', function (event) {
                    appState.filters.prevInst = event.target.value;
                    renderFilteredCards();
                });
            }
        }

        function startDashboardAutoRefresh() {
            if (appState.refreshTimerId) {
                window.clearInterval(appState.refreshTimerId);
            }

            appState.refreshTimerId = window.setInterval(function () {
                loadDashboardData();
            }, appConfig.refreshIntervalMs);
        }

        /**
         * Ponto unico de inicializacao do frontend.
         * Deve ser chamado apenas apos o carregamento completo do documento.
         */
        function initApp() {
            bindFilterEvents();
            loadDashboardData();
            startDashboardAutoRefresh();
        }

        window.addEventListener('load', initApp);
    </script>
